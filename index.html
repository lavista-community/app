UPDATED LOGIN FUNCTION function performLogin() { var email = document.getElementById("emailInput").value; var password = document.getElementById("passwordInput").value; var payload = { action: "login", email: email, password: password }; sendToAppsScript(payload, function(response) { if (response.result === "success") { // Save full profile returned from backend localStorage.setItem("user", JSON.stringify(response.info)); alert("Login successful!"); showDashboard(); } else { alert("Login failed: " + response.message); } }); } // 2. UPDATED DELIVERY REGISTRATION function registerDelivery() { var user = JSON.parse(localStorage.getItem("user")); if (!user) { alert("Please log in first."); return; } var payload = { action: "registerDeliveryPreAuth", surname: user.surname, firstName: user.firstName, address: user.address, riderName: document.getElementById("riderNameInput").value }; sendToAppsScript(payload, function(response) { if (response.result === "success") { alert("Delivery successfully pre-authorized!"); } else { alert("Delivery error: " + response.message); } }); } // 3. CORE COMMUNICATION WRAPPER function sendToAppsScript(payload, callback) { var url = APP_URL + "?callback=handleResponse&payload=" + encodeURIComponent(JSON.stringify(payload)); // Use a global callback function to handle JSONP window.handleResponse = callback; var script = document.createElement('script'); script.src = url; document.body.appendChild(script); } // 4. LOAD DASHBOARD DATA function loadMyWindow() { var user = JSON.parse(localStorage.getItem("user")); var payload = { action: "getMyWindowData", email: user.email, address: user.address }; sendToAppsScript(payload, function(response) { if (response.result === "success") { renderDashboard(response.records); } }); }
